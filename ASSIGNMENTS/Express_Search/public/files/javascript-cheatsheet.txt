  .map(fn)       → new array with transformed values
  .filter(fn)    → new array with matching values
  .reduce(fn, x) → single accumulated value
  .find(fn)      → first match or undefined
  .findIndex(fn) → index of first match or -1
  .some(fn)      → true if any match
  .every(fn)     → true if all match
  .flat(depth)   → flatten nested arrays
  .flatMap(fn)   → map + flat(1)
  .includes(val) → boolean

  Examples:
  const doubled  = [1,2,3].map(n => n * 2);       // [2,4,6]
  const evens    = [1,2,3,4].filter(n => n%2===0); // [2,4]
