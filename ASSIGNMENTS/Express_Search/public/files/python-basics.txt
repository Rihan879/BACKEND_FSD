  # List (mutable, ordered)
  fruits = ["apple", "banana", "cherry"]
  fruits.append("date")
  fruits.remove("apple")
  fruits[0]           # first item
  fruits[-1]          # last item
  fruits[1:3]         # slicing

  # Tuple (immutable, ordered)
  coords = (10, 20)
  x, y = coords       # unpacking

  # Dictionary (key-value pairs)
  person = {"name": "Rihan", "age": 20}
  person["email"] = "r@test.com"    # add/update
  person.get("phone", "N/A")        # safe access
  person.keys()   # dict_keys
  person.values() # dict_values
  person.items()  # dict_items

  # Set (unique, unordered)
  nums = {1, 2, 3, 2, 1}   # → {1, 2, 3}
  nums.add(4)
  nums.discard(1)

  def greet(name, greeting="Hello"):
      return f"{greeting}, {name}!"

  print(greet("Rihan"))           # Hello, Rihan!
  print(greet("Rihan", "Hi"))     # Hi, Rihan!

  # *args and **kwargs
  def total(*numbers):
      return sum(numbers)

  def describe(**info):
      for k, v in info.items():
          print(f"{k}: {v}")

  describe(name="Rihan", age=20, city="Delhi")

  # Lambda

  OBJECT-ORIENTED PROGRAMMING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  class Animal:
      species = "Animal"          # class attribute

      def __init__(self, name, age):
          self.name = name        # instance attribute
          self.age  = age

      def speak(self):
          return f"{self.name} makes a sound."

      def __str__(self):          # string representation
          return f"Animal({self.name}, {self.age})"

      @classmethod
      def create(cls, name):
          return cls(name, 0)

      @staticmethod
      def info():
          return "Animals are living beings."


  class Dog(Animal):              # inheritance
      def speak(self):
          return f"{self.name} says Woof!"

  dog = Dog("Rex", 3)
  print(dog.speak())              # Rex says Woof!
  print(isinstance(dog, Animal))  # True
